export class CreateReservationDto {
  @IsString()
  @IsNotEmpty({ message: 'Le locataire doit être mentionné.' })
  idUser: string;

  @IsDateString()
  @IsNotEmpty({ message: 'La date de début prévue est obligatoire.' })
  dateDebutPrevue: string;

  @IsDateString()
  @IsNotEmpty({ message: 'La date de fin prévue est obligatoire.' })
  dateFinPrevue: string;

  @IsOptional()
  @IsString()
  lieuEvenement?: string;

  @IsOptional()
  @IsString()
  remarques?: string;

  @IsOptional()
  @IsEnum(ModeRetrait)
  modeRetrait?: ModeRetrait;

  @IsOptional()
  @IsNumber()
  @Min(0)
  fraisLivraison?: number;

  @IsIn(['pack', 'personnalise', 'materiel'])
  @IsNotEmpty()
  actionLocataire: 'pack' | 'personnalise' | 'materiel';

  @ValidateIf((o) => o.actionLocataire !== 'materiel')
  @IsString()
  @IsNotEmpty({ message: 'codePack requis pour un pack ou une personnalisation.' })
  codePack?: string;

  @ValidateIf((o) => o.actionLocataire === 'pack')
  @IsOptional()
  @IsInt()
  @Min(1)
  quantitePack?: number;

  @ValidateIf((o) => o.actionLocataire !== 'pack')
  @IsArray()
  @ArrayMinSize(1, { message: 'Au moins un matériel est requis.' })
  @ValidateNested({ each: true })
  @Type(() => LigneMaterielDto)
  materiels?: LigneMaterielDto[];
}

 
  export class LigneMaterielDto {
    @IsString()
    @IsNotEmpty({ message: 'Le code matériel est obligatoire.' })
    codeMateriel: string;
  
    @IsInt({ message: 'La quantité doit être un entier.' })
    @Min(1, { message: 'La quantité doit être d\'au moins 1.' })
    quantite: number;
  
    @IsOptional()
    @IsNumber()
    @Min(0)
    tarifApplique?: number;
  }
